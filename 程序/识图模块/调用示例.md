## 一、装配

```js
const { createVisionModule } = require('./识图模块/index.js');
const { createRealAdapter } = require('./识图模块/adapter-real.js');

const 端口 = createRealAdapter({
  取图字节: (引用) => {
    const 回 = 图片.run('取内容', { 引用 });
    return 回 && 回.ok ? 回.data : null;      // { 字节, mime, 文件名 }
  },
  查词: (词们) => 标签.recognize(词们),
  tagger: 引擎.打标,                          // 对接层/模型引擎/识图.js
  读元数据: 读PNG元数据,                        // 识图模块/元数据.js
  看图: 造看图({ 调度, 请求名: '识图', 提示词 }),
  引擎: { id: 引擎.id, 名字: 引擎.名字, 阈值: 引擎.阈值 },
});

const 识图 = createVisionModule(端口);
```

按引用读取需要 `取图字节`；直接传字节读元数据只需要解析端口：

```js
const 元数据识图 = createVisionModule({ 读元数据: 读PNG元数据 });
await 元数据识图.run('读元数据', { 字节: pngBytes });
```

## 二、读元数据（先试这条）

```js
const 回 = await 识图.run('读元数据', { 存放地: 'gallery', 条目ID: 'img_ab12' });

// 界面刚收到一份原始 PNG 时可直接传字节，不需要先存进图片模块
const 直接读 = await 识图.run('读元数据', { 字节: pngBytes });

// AI 画的图，回
{
  ok: true,
  data: {
    引用: { 作用域: { 存放地: 'gallery' }, 条目ID: 'img_ab12' },   // 回执里仍然给收好的那一份
    有: true,
    提示词: 'a cat, sitting',
    负向提示词: 'blurry, lowres',
    工作流: '{"1":{"class_type":"KSampler"}}',
    参数: 'a cat, sitting\nNegative prompt: blurry, lowres\nSteps: 20, Sampler: Euler',
    全部: { parameters: '…', workflow: '…', Software: 'SomeTool' },
    尺寸: { 宽: 8, 高: 8 },
  },
}
```

直接传字节时，回执的 `引用` 为 `null`；按引用读取时仍回原有的收好引用。

**不是 PNG 或没给这一支**：`{ ok:false, code:'VISION_BAD_METADATA', reason:'VISION_BAD_METADATA: 不是 PNG' }`

const 回 = await 识图.run('认图', { 存放地: 'gallery', 条目ID: 'img_ab12', 阈值: 0.6, 上限: 20 });

  ok: true,

## 四、看图（交给模型）

```js
const 回 = await 识图.run('看图', {
  存放地: 'gallery',
  条目ID: 'img_ab12',
  提示: '这是什么风格，画里有几个人',
  参考标签: ['1girl', 'solo'],          // 通常就是认图出来的那批
});

// 回
{ ok:true, data:{ 引用, 标签:[…], 描述:'一个女孩站在窗边，动画风格', 模型:'deepseek-flash', 用量:{…} } }
```

**没接看图链路**：`{ ok:false, code:'VISION_NO_VIEWER', reason:'VISION_NO_VIEWER: 没有看图的链路' }`
**链路抛错**：`{ ok:false, code:'VISION_VIEW_FAILED', reason:'VISION_VIEW_FAILED: 模型拒了这张图' }`

## 五、看这一格能不能用

```js
识图.run('看模型');
// { ok:true, data:{ 可用:true, 缺:[], 能本地打标:true, 能读元数据:true, 能看图:true,
//                    有查词:true, 模型:{ id:'wd-eva02', 名字:'WD EVA02 Tagger' } } }
```

**它只报端口齐不齐**，不去探盘上有没有模型文件——"有没有"是装配层与界面的事。

## 六、三条路怎么配合

```js
// 先看文件里有没有现成的
const 元 = await 识图.run('读元数据', { 存放地: 'gallery', 条目ID: 'img_ab12' });
if (元.ok && 元.data.提示词) return 元.data.提示词;

// 没有就问本地模型
const 认 = await 识图.run('认图', { 存放地: 'gallery', 条目ID: 'img_ab12' });
const 参考 = 认.ok ? 认.data.标签.map((一) => 一.词) : [];

// 还不够细就交给模型
const 看 = await 识图.run('看图', { 存放地: 'gallery', 条目ID: 'img_ab12', 提示: '描述这张图', 参考标签: 参考 });
return 看.ok ? 看.data.描述 : '';
```

## 七、错误表

| 码 | 什么时候 | 界面怎么办 |
|---|---|---|
| `VISION_BAD_REF` | 存放地 缺了 / 不在那三个里，或 条目ID 是空的 | 让调用方补上（模型常常会漏） |
| `VISION_NO_BYTES` | 图不在、取图那一支抛错、没给取图端口 | 说"这张图取不到"，别当成模型的问题 |
| `VISION_NO_ENGINE` | 没递本地打标 | 说"本地识图未配置" |
| `VISION_ENGINE_FAILED` | 打标引擎抛错 | `reason` 里有引擎报的原话 |
| `VISION_NO_VIEWER` | 没递看图链路 | 说"看图未配置" |
| `VISION_VIEW_FAILED` | 看图链路抛错 | `reason` 里有链路报的原话 |
| `VISION_BAD_METADATA` | 不是 PNG / 没给读元数据那一支 | 说"这张图里没有可读的生成信息" |
| `VISION_BAD_ARGUMENT` | 动作名不对 / 动作自己抛错 | 改代码 |
